% Motivation Section
\begin{block}{\textbf{Motivation}}
\begin{itemize}
\justifying
    \item Digital subscription platforms run on \textbf{recurring revenue}: keeping existing subscribers is far cheaper than acquiring new ones, and usage is both an antecedent and a consequence of satisfaction \cite{bolton1999}.
    \item Platforms nevertheless often treat all subscribers identically, deploying \textbf{generic retention campaigns} even though behavioural signals separate engaged readers from those at risk \cite{coussement2008,vuorinen2024}.
    \item \textbf{Research question:} which behavioural and demographic signals distinguish high-engagement from low-engagement eBook subscribers, and what should the platform do about each group?
    \item \textbf{Phase 1} (customer file): a subscription-propensity view of 16,404 unique customers (115 duplicate IDs removed), 33.2\% subscribers. \textbf{Phase 2} (engagement file): score 5,455 subscribers and assign an Active / Inactive / Churned status.
\end{itemize}
\end{block}

% Key Research Contribution highlighted
\begin{alertblock}{\textbf{Key Contributions}}

\begin{itemize}
\justifying
\item A reproducible \textbf{0--100 engagement score} with a one-line justification for every weight (0.40 recency, 0.35 logins, 0.25 session minutes) and \textbf{no double-counting}: Books Accessed is excluded because it correlates $r \approx 0.88$ with logins.

\item \textbf{Status rules written and approved before computation}: Active $\le$ 30 days since last activity, Inactive 31--90, Churned $>$ 90 -- each anchored to a 30-day subscription cycle, not learned from the data.

\item An \textbf{interactive dashboard prototype} (Streamlit) with per-segment recommendations tied to the literature: retain Active, win back Inactive before day 90, and a win-back campaign for Churned \cite{thomas2004,gomezuribe2016}.
\end{itemize}

\end{alertblock}

% Experimental Setup and Training protocol
\begin{block}{\textbf{Experimental Setup \& Protocol}}

\begin{itemize}
\justifying
\item \textbf{Dataset:} engagement file (5,500 rows; 45 duplicates removed $\rightarrow$ 5,455 subscribers) joined on Customer ID to the customer file (16,519 rows; 16,404 unique customers; 33.2\% subscribers).

\item \textbf{Cleaning (C1--C10):} 26 raw genre spellings mapped to 8 genres; implausible logins / session minutes / books blanked; invalid or missing recency \textbf{never imputed} (33 subscribers stay Unclassified); 144 blank logins / session values median-filled.

\item \textbf{Score:} $100 \times (0.40 \times \text{inverted recency} + 0.35 \times \text{logins} + 0.25 \times \text{session minutes})$, min--max scaled -- recency is the most direct and explainable signal of leaving, logins measure usage frequency, session minutes depth of use. Terciles: 40.80 / 51.25.

\item \textbf{Status rules:} Active $\le$ 30 days and $\ge$ 1 login/month; Inactive 31--90 days; Churned $>$ 90 days.

\item \textbf{Validation:} Books Accessed (not a score input) by tier; logins by status; demographic effect sizes. \textbf{Tools:} Python, pandas, NumPy, Matplotlib, Seaborn, Streamlit.

\end{itemize}

\end{block}

\begin{itemize}
\justifying
    \item \textbf{Demographic differences are small}: high-value subscribers (Active + High tier) vs Churned differ by \$94.1 vs \$83.5 monthly spend ($d = 0.33$) and \$99.8k vs \$90.3k income ($d = 0.23$) -- significant, but small effects; associations, not causes.
    \item \textbf{Genre and tenure do not separate groups}: Active rates span only 52--58\% across genres, and churn is non-monotone across tenure bands -- reported as null findings.
    \item \textbf{Tier--status agreement is partly by construction}: recency drives both the status rule and 40\% of the score; the books panel above is the independent evidence.
\end{itemize}

\begin{alertblock}{\textbf{Conclusion}}

\begin{itemize}
\justifying
    \item Declining \textbf{login frequency is an early-warning signal}: future Churners still averaged 1.5 logins/month before falling silent for a median 152 days \cite{coussement2008,bolton1999}.
    \item \textbf{Usage intensity, not demographics, separates segments} -- segment-specific strategies beat generic campaigns, and targeting must be calibrated to avoid wasteful outreach \cite{ascarza2018}.
    \item \textbf{Personalised, behaviour-triggered win-back} beats blanket offers; sustained engagement follows content-consumption patterns \cite{gomezuribe2016,thomas2004,godinho2020,rani2023}.
    \item \textbf{Recommendations:} \emph{Active} -- retain with personalised recommendations and watch the \textbf{11 Active subscribers already in the Low tier} (early-warning list); \emph{Inactive} -- win back before day 90 (median 58 days quiet); \emph{Churned} -- targeted win-back campaign.
\end{itemize}

\end{alertblock}

\begin{block}{\textbf{Limitations \& Future Work}}
\begin{itemize}
\justifying
    \item Shows \textbf{who} leaves and \textbf{when}, not \textbf{why}: no cancellation reasons or dates, so tenure views are cross-sectional, not retention curves.
    \item Status is recency-based by definition; only the single most-read genre is recorded; 33 subscribers remain Unclassified; the dataset is synthetic, so findings illustrate the method, not market claims.
    \item \textbf{Future:} pilot the three strategies, extend with real cancellation data, and add the Phase~1 propensity model as a second decision lens.
\end{itemize}
\end{block}
